const firebaseConfig = { apiKey: "YOUR_API_KEY", // Replace with your Web API Key from the Firebase Console authDomain: "trailman100-membersip-system.firebaseapp.com", projectId: "trailman100-membersip-system" }; const app = initializeApp(firebaseConfig); const auth = getAuth(app); const db = getFirestore(app); // Actively monitor the user session onAuthStateChanged(auth, async (user) => { if (!user) { // Kick unauthenticated traffic back to login window.location.href = 'index.html'; } else { try { // Query Firestore using the authenticated user's UID const docRef = doc(db, "members", user.uid); const docSnap = await getDoc(docRef); document.getElementById('loading').style.display = 'none'; document.getElementById('profile').style.display = 'block'; document.getElementById('memberEmail').innerText = user.email; if (docSnap.exists()) { const data = docSnap.data(); document.getElementById('memberName').innerText = data.fullName || 'No name provided'; document.getElementById('memberLevel').innerText = data.membershipLevel || 'Standard'; } else { document.getElementById('memberName').innerText = 'No profile found'; document.getElementById('memberLevel').innerText = 'N/A'; } } catch (err) { console.error("Error grabbing Firestore profile document: ", err); } } }); // Sign out functionality document.getElementById('logoutBtn').addEventListener('click', () => { signOut(auth).then(() => { window.location.href = 'index.html'; }); });
